import os
import logging
from contextlib import asynccontextmanager
from fastapi import FastAPI, HTTPException
from fastapi.responses import JSONResponse
# ดึง engine มาจากไฟล์ที่คุณเพิ่งทำเสร็จ
from decision_engine import engine

# 1. ตั้งค่า Logger (ก่อน lifespan จะได้อ้างอิงได้)
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger("AlphaBuffaloGateway")

# 2. ใช้ lifespan แทน on_event (มาตรฐานใหม่)
@asynccontextmanager
async def lifespan(app: FastAPI):
    logger.info("Gateway Started. System Ready.")
    yield
    logger.info("Gateway Shutdown.")

app = FastAPI(lifespan=lifespan)

# 3. จุดเชื่อมต่อจริง (เชื่อมกับ decision_engine)
@app.post("/webhook")
async def handle_signal(data: dict):
    try:
        # ส่งค่าเข้า Engine V12 ของคุณ
        decision = engine.run(
            fibo=float(data.get("fibo", 0)),
            score=float(data.get("score", 0)),
            volatility_score=float(data.get("vol", 0))
        )
        return {"allowed": decision.allowed, "confidence": decision.confidence, "reason": decision.reason}
    except Exception as e:
        logger.error(f"Engine Error: {e}")
        return JSONResponse(status_code=500, content={"status": "error", "message": str(e)})

if __name__ == "__main__":
    import uvicorn
    port = int(os.getenv("PORT", 8080))
    uvicorn.run(app, host="0.0.0.0", port=port)
